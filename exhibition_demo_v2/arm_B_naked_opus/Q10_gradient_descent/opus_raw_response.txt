```python
from manim import *
import numpy as np


class MainScene(Scene):
    def construct(self):
        # ---------- Title ----------
        title = Text("Gradient Descent", font_size=48, color=YELLOW)
        subtitle = MathTex(r"f(x,y) = x^2 + 2y^2", font_size=40)
        subtitle.next_to(title, DOWN)
        self.play(Write(title), FadeIn(subtitle, shift=UP))
        self.wait(1)
        self.play(
            title.animate.scale(0.55).to_edge(UP).shift(LEFT * 3.5),
            subtitle.animate.scale(0.7).to_edge(UP).shift(RIGHT * 2.5),
        )
        self.wait(0.3)

        # ---------- Axes ----------
        axes = Axes(
            x_range=[-4, 4, 1],
            y_range=[-3, 3, 1],
            x_length=7,
            y_length=5.2,
            axis_config={"include_tip": True, "color": GREY_B},
        ).to_edge(DOWN, buff=0.4).shift(LEFT * 2.2)
        self.play(Create(axes))

        # ---------- Contour ellipses ----------
        def f(x, y):
            return x ** 2 + 2 * y ** 2

        contours = VGroup()
        levels = [1, 4, 9, 16, 25]
        colors = color_gradient([BLUE_E, TEAL, GREEN], len(levels))
        for c, col in zip(levels, colors):
            a = np.sqrt(c)          # x semi-axis
            b = np.sqrt(c / 2)      # y semi-axis
            ell = ParametricFunction(
                lambda t, a=a, b=b: axes.c2p(a * np.cos(t), b * np.sin(t)),
                t_range=[0, TAU],
                color=col,
                stroke_width=3,
            )
            contours.add(ell)
        self.play(LaggedStartMap(Create, contours, lag_ratio=0.2), run_time=2.5)
        clabel = Text("Contour lines of f", font_size=22, color=TEAL)
        clabel.next_to(axes, DOWN, buff=0.15)
        self.play(FadeIn(clabel))
        self.wait(0.5)

        # ---------- Gradient formula ----------
        grad_box = MathTex(r"\nabla f = (2x,\ 4y)", font_size=34, color=ORANGE)
        grad_box.to_edge(RIGHT, buff=0.5).shift(UP * 2)
        self.play(Write(grad_box))
        self.wait(0.5)

        # ---------- Start point ----------
        lr = 0.1
        point = np.array([3.0, 2.0])
        dot = Dot(axes.c2p(*point), color=RED, radius=0.09)
        dot_lbl = MathTex(r"(3, 2)", font_size=26, color=RED).next_to(dot, UR, buff=0.1)
        self.play(FadeIn(dot, scale=2), Write(dot_lbl))
        self.wait(0.4)

        # ---------- Descent direction ----------
        grad = np.array([2 * point[0], 4 * point[1]])  # (6, 8)
        neg = -grad  # (-6, -8)
        arrow = Arrow(
            axes.c2p(*point),
            axes.c2p(*(point + lr * neg)),
            color=RED,
            buff=0,
            stroke_width=6,
            max_tip_length_to_length_ratio=0.3,
        )
        step_calc = VGroup(
            MathTex(r"-\nabla f = (-6, -8)", font_size=30, color=ORANGE),
            MathTex(r"\eta = 0.1", font_size=30, color=WHITE),
            MathTex(r"x_{1} = (2.4,\ 1.2)", font_size=30, color=YELLOW),
        ).arrange(DOWN, aligned_edge=LEFT).next_to(grad_box, DOWN, buff=0.5)
        self.play(GrowArrow(arrow), Write(step_calc[0]))
        self.play(Write(step_calc[1]))
        self.wait(0.3)

        # ---------- First step ----------
        new_point = point + lr * neg
        new_dot = Dot(axes.c2p(*new_point), color=YELLOW, radius=0.08)
        path = VMobject(color=RED, stroke_width=4)
        path.set_points_as_corners([axes.c2p(*point), axes.c2p(*new_point)])
        self.play(
            Write(step_calc[2]),
            Transform(dot.copy(), new_dot),
            Create(path),
            FadeOut(arrow),
        )
        self.add(new_dot)
        self.wait(0.4)

        iter_lbl = Text("Iterating...", font_size=24, color=GREEN).next_to(step_calc, DOWN, buff=0.5)
        self.play(FadeIn(iter_lbl))

        # ---------- 8 more iterations ----------
        cur = new_point.copy()
        all_points = [point, new_point]
        for i in range(8):
            g = np.array([2 * cur[0], 4 * cur[1]])
            nxt = cur - lr * g
            seg = Line(axes.c2p(*cur), axes.c2p(*nxt), color=RED, stroke_width=4)
            d = Dot(axes.c2p(*nxt), color=YELLOW, radius=0.06)
            self.play(Create(seg), FadeIn(d), run_time=0.35)
            cur = nxt
            all_points.append(nxt)
        self.wait(0.4)

        # ---------- Convergence insight ----------
        self.play(FadeOut(iter_lbl), FadeOut(step_calc))
        insight = VGroup(
            Text("y converges faster than x", font_size=24, color=GREEN_B),
            MathTex(r"\text{coeff } y\,(4) > \text{coeff } x\,(2)", font_size=26, color=GREEN_B),
        ).arrange(DOWN, buff=0.25).next_to(grad_box, DOWN, buff=0.6)
        self.play(Write(insight))
        self.wait(1)

        # ---------- Final point ----------
        final_dot = Dot(axes.c2p(*cur), color=GREEN, radius=0.1)
        final_lbl = MathTex(r"\nabla f \approx 0", font_size=28, color=GREEN).next_to(final_dot, UR, buff=0.15)
        pulse = Circle(radius=0.3, color=GREEN).move_to(axes.c2p(*cur))
        self.play(FadeIn(final_dot, scale=2), Write(final_lbl))
        self.play(Create(pulse), pulse.animate.scale(0.1).set_opacity(0), run_time=1.2)
        self.wait(2)
```
